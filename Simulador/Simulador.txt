    <summary>Cómo funciona el modelo (para el informe)</summary>
    <h3>Entidades y recursos</h3>
    <ul>
      <li><b>Pedidos</b>: tienen número de factura, tipo (OUT o POP) y bultos (distribución triangular entre 3, 12 y el máximo configurado).</li>
      <li><b>Chequeadores</b> y <b>cargadores</b>: son los recursos que atienden. Cada uno camina a una velocidad configurable (1 píxel = 0,1 m).</li>
      <li><b>Estivas</b>: almacenan los pedidos ubicados. Si no hay espacio, se asigna igual la menos ocupada y se cuenta un exceso, igual que en el sistema real (avisa, pero no bloquea).</li>
      <li><b>Camión</b>: tiene capacidad en bultos. Sale cuando está lleno o cuando ningún pedido ubicado cabe en lo que le queda; después hay un tiempo de cambio hasta que llega el siguiente.</li>
    </ul>
    <h3>Proceso</h3>
    <ul>
      <li>Llegan pedidos a recepción con tiempos entre llegadas exponenciales (proceso de Poisson).</li>
      <li>Un chequeador libre toma el primero de la cola (FIFO). El chequeo dura una variable normal truncada (σ = 25 % de la media).</li>
      <li>El chequeador lo lleva a una estiva según la política de ubicación y lo coloca. Luego vuelve a su puesto.</li>
      <li>Un cargador libre elige un pedido ubicado según la política de carga, lo recoge, lo lleva al camión y lo carga. El tiempo de carga es proporcional a los bultos.</li>
    </ul>
    <h3>Cómo se simula el tiempo</h3>
    <ul>
      <li>Es una simulación basada en agentes con reloj discreto (Δt = 0,5 s). Las llegadas, los tiempos de chequeo y los bultos son aleatorios, con una semilla fija para poder repetir los experimentos.</li>
    </ul>
    <h3>Supuestos a validar</h3>
    <ul>
      <li>Los valores por defecto son <b>supuestos de ejemplo</b>, no mediciones. Para el informe conviene reemplazarlos por tiempos reales cronometrados en el almacén (llegadas por hora, tiempo de chequeo, caminata y carga por bulto, capacidad real del camión).</li>
      <li>Las rutas siguen pasillos fijos y las personas no se bloquean entre sí.</li>

function mulberry32(a) {
  return function () {
    a |= 0; a = a + 0x6D2B79F5 | 0;
    let t = Math.imul(a ^ a >>> 15, 1 | a);
    t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
    return ((t ^ t >>> 14) >>> 0) / 4294967296;
  };
}

const expo = (media) => -media * Math.log(1 - rand());
function normal(media, sd) {
  const u = 1 - rand(), v = rand();
  return media + sd * Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * v);
}
function tri(a, c, b) {
  const u = rand(), F = (c - a) / (b - a);
  return u < F ? a + Math.sqrt(u * (b - a) * (c - a)) : b - Math.sqrt((1 - u) * (b - a) * (b - c));
}
const pad = (n, l) => String(n).padStart(l, '0');

  t = Math.floor(t);

}

function addLog(msg) {

const fac = (p) => 'F-' + pad(p.id, 4);

/* ====================== lógica ====================== */
function mover(w, dt, vel) {
  let d = vel * dt;
  while (d > 0 && w.path.length) {
    const p = w.path[0], dx = p[0] - w.x, dy = p[1] - w.y, dist = Math.hypot(dx, dy);
    if (dist <= d) { w.x = p[0]; w.y = p[1]; w.path.shift(); d -= dist; }
    else { w.x += dx / dist * d; w.y += dy / dist * d; d = 0; }
  }
  return w.path.length === 0;
}

function crearPedido(tLleg) {
  const b = Math.max(1, Math.round(tri(3, 12, cfg.bMax)));
  const p = { id: S.nextId++, tipo: rand() < cfg.pPop ? 'POP' : 'OUT', bultos: b, estado: 'cola', tLleg,
              tIniChq: null, tUbic: null, tIniCarga: null, tCarg: null, estiva: null, camion: null, reservado: false };
  S.pedidos.push(p);
  S.cola.push(p);
  S.st.maxCola = Math.max(S.st.maxCola, S.cola.length);
  addLog('Llega ' + fac(p) + ' (' + b + ' bultos, ' + p.tipo + ') a recepción.');
}

function elegirEstiva(p) {
  const cand = S.estivas.filter((e) => e.ocup + e.resIn + p.bultos <= cfg.capEst);
  const menos = (arr) => arr.reduce((a, b) => (a.ocup + a.resIn <= b.ocup + b.resIn ? a : b));
  if (!cand.length) { return menos(S.estivas); }
  if (cfg.polUbi === 'primera') { return cand[0]; }
  if (cfg.polUbi === 'menos') { return menos(cand); }
  return cand[Math.floor(rand() * cand.length)];
}

function actualizarChq(w) {
  const vel = cfg.vel * 10;
  switch (w.state) {
    case 'libre':
      if (S.cola.length) {
        const p = S.cola.shift();
        p.estado = 'chequeando'; p.tIniChq = S.t;
        S.st.sumEspRec += S.t - p.tLleg; S.st.nEspRec++;
        w.pedido = p;
        w.total = w.timer = Math.max(20, normal(cfg.tChq, 0.25 * cfg.tChq));
        w.state = 'chequeando';
        addLog('Chequeador ' + w.n + ' empieza a chequear ' + fac(p) + '.');
      }
      break;
    case 'chequeando':
      w.timer -= DT;
      if (w.timer <= 0) {
        const e = elegirEstiva(w.pedido);
        w.estiva = e; e.resIn += w.pedido.bultos;
        w.pedido.estado = 'a_estiva';
        w.path = [[CORR_L, w.hy], [CORR_L, e.ay], [e.ax, e.ay]];
        w.state = 'camino'; w.timer = 0;
        addLog('Chequeador ' + w.n + ' lleva ' + fac(w.pedido) + ' a la estiva E' + e.id + '.');
      }
      break;
    case 'camino':
      if (mover(w, DT, vel)) { w.state = 'colocando'; w.total = w.timer = 3 + 0.3 * w.pedido.bultos; }
      break;
    case 'colocando':
      w.timer -= DT;
      if (w.timer <= 0) {
        const e = w.estiva, p = w.pedido;
        e.resIn -= p.bultos; e.ocup += p.bultos; e.pedidos.push(p);
        p.estado = 'ubicado'; p.tUbic = S.t; p.estiva = e.id;
        const exc = e.ocup > cfg.capEst;
        if (exc) { S.st.excesos++; }
        addLog(fac(p) + ' ubicado en E' + e.id + ' (' + e.ocup + '/' + cfg.capEst + ')' + (exc ? ' ⚠ excede la capacidad' : '') + '.');
        w.pedido = null; w.timer = 0;
        w.path = [[CORR_L, e.ay], [CORR_L, w.hy], [w.hx, w.hy]];
        w.state = 'regreso';
      }
      break;
    case 'regreso':
      if (mover(w, DT, vel)) { w.state = 'libre'; w.estiva = null; }
      break;
  }
  if (w.state !== 'libre') { w.busy += DT; }
}

function elegirParaCargar() {
  const T = S.truck;
  const restante = cfg.capCam - T.carga - T.res;
  let mejor = null, mejorE = null, mejorClave = null;
  for (const e of S.estivas) {
    for (const p of e.pedidos) {
      if (p.reservado || p.bultos > restante) { continue; }
      let clave;
      if (cfg.polCar === 'fifo') { clave = p.tUbic; }
      else if (cfg.polCar === 'cercano') { clave = Math.hypot(e.ax - TRUCK_REAR_X, e.ay - TRUCK_Y); }
      else { clave = -p.bultos; }
      if (mejorClave === null || clave < mejorClave) { mejorClave = clave; mejor = p; mejorE = e; }
    }
  }
  return mejor ? { p: mejor, e: mejorE } : null;
}

function actualizarCar(w) {
  const vel = cfg.vel * 10, T = S.truck;
  switch (w.state) {
    case 'libre':
      if (T.estado === 'presente' && !T.manual) {
        const sel = elegirParaCargar();
        if (sel) {
          const p = sel.p, e = sel.e;
          p.reservado = true; p.tIniCarga = S.t; p.estado = 'a_recoger';
          S.st.sumEspEst += S.t - p.tUbic; S.st.nEspEst++;
          T.res += p.bultos;
          w.pedido = p; w.estiva = e;
          w.path = [[CORR_R, w.hy], [CORR_R, e.ay], [e.ax, e.ay]];
          w.state = 'camino_estiva';
          addLog('Cargador ' + w.n + ' va por ' + fac(p) + ' a la estiva E' + e.id + '.');
        }
      }
      break;
    case 'camino_estiva':
      if (mover(w, DT, vel)) { w.state = 'recogiendo'; w.total = w.timer = 3 + 0.5 * w.pedido.bultos; }
      break;
    case 'recogiendo':
      w.timer -= DT;
      if (w.timer <= 0) {
        const e = w.estiva, p = w.pedido;
        e.pedidos.splice(e.pedidos.indexOf(p), 1);
        e.ocup -= p.bultos;
        p.estado = 'a_camion';
        w.path = [[CORR_R, e.ay], [CORR_R, TRUCK_Y], [TRUCK_REAR_X, TRUCK_Y]];
        w.state = 'camino_camion'; w.timer = 0;
      }
      break;
    case 'camino_camion':
      if (mover(w, DT, vel)) { w.state = 'cargando'; w.total = w.timer = 2 + cfg.sBulto * w.pedido.bultos; }
      break;
    case 'cargando':
      w.timer -= DT;
      if (w.timer <= 0) {
        const p = w.pedido;
        T.carga += p.bultos; T.res -= p.bultos;
        p.estado = 'cargado'; p.tCarg = S.t; p.camion = T.nro;
        S.st.sumSis += S.t - p.tLleg; S.st.nSis++; S.st.cargados++; S.st.bultosCargados += p.bultos;
        addLog('Cargador ' + w.n + ' carga ' + fac(p) + ' al camión Nº ' + T.nro + ' (' + T.carga + '/' + cfg.capCam + ').');
        w.pedido = null; w.timer = 0;
        w.path = [[CORR_R, TRUCK_Y], [CORR_R, w.hy], [w.hx, w.hy]];
        w.state = 'regreso';
      }
      break;
    case 'regreso':
      if (mover(w, DT, vel)) { w.state = 'libre'; w.estiva = null; }
      break;
  }
  if (w.state !== 'libre') { w.busy += DT; }
}

function despachar() {
  const T = S.truck;
  T.estado = 'saliendo'; T.manual = false;
  S.st.camiones++; S.st.bultosDesp += T.carga;
  addLog('Camión Nº ' + T.nro + ' sale con ' + T.carga + ' bultos (' + Math.round(T.carga / cfg.capCam * 100) + ' % de su capacidad).');
}

function actualizarCamion() {
  const T = S.truck, vmov = 280 / TRUCK_MOVE;
  switch (T.estado) {
    case 'presente': {
      if (T.res === 0 && T.carga > 0) {
        const restante = cfg.capCam - T.carga;
        const hayUbic = S.estivas.some((e) => e.pedidos.length > 0);
        const cabe = S.estivas.some((e) => e.pedidos.some((p) => p.bultos <= restante));
        if (restante <= 0 || (hayUbic && !cabe) || T.manual) { despachar(); }
      }
      break;
    }
    case 'saliendo':
      T.off += vmov * DT;
      if (T.off >= 280) { T.off = 280; T.estado = 'cambio'; T.espera = cfg.tCambio; }
      break;
    case 'cambio':
      T.espera -= DT;
      if (T.espera <= 0) {
        T.nro++; T.carga = 0; T.res = 0; T.estado = 'llegando';
        addLog('Llega el camión Nº ' + T.nro + ' al muelle.');
      }
      break;
    case 'llegando':
      T.off -= vmov * DT;
      if (T.off <= 0) { T.off = 0; T.estado = 'presente'; }
      break;
  }
}

function contarUbicados() {
  let n = 0, b = 0;
  S.estivas.forEach((e) => { n += e.pedidos.length; b += e.ocup; });
  return { n, b };
}

function paso() {
  S.t += DT;
  while (S.t >= S.nextArrival) {
    const t = S.nextArrival;
    crearPedido(t);
    S.nextArrival += expo(3600 / cfg.rate);
  }
  S.chq.forEach(actualizarChq);
  actualizarCamion();
  S.car.forEach(actualizarCar);
  S.st.maxCola = Math.max(S.st.maxCola, S.cola.length);

  if (S.t >= S.nextSample) {
    const u = contarUbicados();
    S.serie.push({ t: S.t, cola: S.cola.length, ubic: u.n });
    S.nextSample += 30;
  }
  if (cfg.jornada > 0 && S.t >= cfg.jornada) { terminar(); }
}

function terminar() {
  S.fin = true; running = false;
  $('btnPlay').textContent = '▶ Iniciar';
  const m = metricas();
  $('finTxt').textContent = m.cargados + ' pedidos cargados (' + m.bultosCargados + ' bultos) en ' + m.camiones +
    ' camiones despachados. Tiempo medio en el sistema: ' + min1(m.tSis) + ' min. Cola máxima en recepción: ' + S.st.maxCola + '.';
  $('fin').style.display = 'flex';
  refrescarUI(true);
}

/* ====================== métricas ====================== */
function metricas() {
  const st = S.st, u = contarUbicados();
  const cnt = { cola: S.cola.length, chq: 0, aEst: 0, aCam: 0 };
  S.pedidos.forEach((p) => {
    if (p.estado === 'chequeando') { cnt.chq++; }
    else if (p.estado === 'a_estiva') { cnt.aEst++; }
    else if (p.estado === 'a_recoger' || p.estado === 'a_camion') { cnt.aCam++; }
  });
  const t = Math.max(S.t, 1);
  const busyC = S.chq.reduce((a, w) => a + w.busy, 0), busyK = S.car.reduce((a, w) => a + w.busy, 0);
  return {
    llegados: S.pedidos.length, cola: cnt.cola, chq: cnt.chq, aEst: cnt.aEst, aCam: cnt.aCam,
    ubicN: u.n, ubicB: u.b, cargados: st.cargados, bultosCargados: st.bultosCargados, camiones: st.camiones,
    bultosDesp: st.bultosDesp, tRec: st.nEspRec ? st.sumEspRec / st.nEspRec : 0,
    tEst: st.nEspEst ? st.sumEspEst / st.nEspEst : 0, tSis: st.nSis ? st.sumSis / st.nSis : 0,
    utilChq: busyC / (t * cfg.nChq) * 100, utilCar: busyK / (t * cfg.nCar) * 100,
    porHora: st.cargados / (t / 3600), excesos: st.excesos
  };
}

function dibujarGrafico() {
  const cv = $('chart'), g = cv.getContext('2d'), w = cv.width, h = cv.height;
  g.clearRect(0, 0, w, h);
  const ml = 38, mb = 24, mt = 26, mr = 8;
  const tmax = cfg.jornada > 0 ? cfg.jornada : Math.max(S.t, 3600);
  const ymax = Math.max(5, ...S.serie.map((s) => Math.max(s.cola, s.ubic)));
  g.strokeStyle = '#33424d'; g.fillStyle = '#8fa0ad'; g.font = '13px sans-serif'; g.lineWidth = 1;
  for (let i = 0; i <= 4; i++) {
    const y = mt + (h - mt - mb) * (1 - i / 4);
    g.beginPath(); g.moveTo(ml, y); g.lineTo(w - mr, y); g.stroke();
    g.fillText(String(Math.round(ymax * i / 4)), 4, y + 4);
  }
  const nx = cfg.jornada > 0 ? cfg.jornada / 3600 : Math.ceil(tmax / 3600);
  for (let i = 0; i <= nx; i++) {
    const x = ml + (w - ml - mr) * (i * 3600 / tmax);
    if (x <= w - mr + 1) { g.fillText(i + ' h', x - 8, h - 6); }
  }
  const serie = (clave, color) => {
    g.strokeStyle = color; g.lineWidth = 2; g.beginPath();
    S.serie.forEach((s, i) => {
      const x = ml + (w - ml - mr) * (s.t / tmax), y = mt + (h - mt - mb) * (1 - s[clave] / ymax);
      if (i === 0) { g.moveTo(x, y); } else { g.lineTo(x, y); }
    });
    g.stroke();
  };
  serie('cola', '#e0a030');
  serie('ubic', '#00a8c6');
  g.fillStyle = '#e0a030'; g.fillRect(ml, 6, 12, 5); g.fillStyle = '#cbd5dd'; g.fillText('En recepción', ml + 18, 13);
  g.fillStyle = '#00a8c6'; g.fillRect(ml + 150, 6, 12, 5); g.fillStyle = '#cbd5dd'; g.fillText('Ubicados en estivas', ml + 168, 13);
}

    'Pedido;Tipo;Bultos;Llegada (s);Inicio chequeo (s);Ubicado (s);Inicio carga (s);Cargado (s);Estiva;Camión'
  ];
  S.pedidos.forEach((p) => {
    lineas.push([fac(p), p.tipo, p.bultos, f(p.tLleg), f(p.tIniChq), f(p.tUbic), f(p.tIniCarga), f(p.tCarg), p.estiva === null ? '' : 'E' + p.estiva, p.camion === null ? '' : p.camion].join(';'));
  });
  const blob = new Blob(['﻿' + lineas.join('\r\n')], { type: 'text/csv;charset=utf-8' });
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob); a.download = 'simulacion_almacen.csv';
  document.body.appendChild(a); a.click(); a.remove();
  setTimeout(() => URL.revokeObjectURL(a.href), 1000);
}

let ultimo = 0, acum = 0;
function frame(ahora) {
  const dt = Math.min(0.1, (ahora - ultimo) / 1000);
  ultimo = ahora;
  if (running && !S.fin) {
    acum += dt * (+$('speed').value);
    let n = 0;
    while (acum >= DT && n < 4000 && !S.fin) { paso(); acum -= DT; n++; }
    if (n >= 4000) { acum = 0; }
  }
  dibujar();
  refrescarUI(false);
  requestAnimationFrame(frame);
}

  if (S.fin) { return; }

}
