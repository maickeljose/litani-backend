DROP DATABASE IF EXISTS litani_db;
CREATE DATABASE litani_db CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE litani_db;

-- ---------- ROLES ----------
CREATE TABLE roles (
  id_rol     INT NOT NULL AUTO_INCREMENT,
  nombre_rol VARCHAR(50) NOT NULL,
  PRIMARY KEY (id_rol),
  UNIQUE KEY uq_roles_nombre (nombre_rol)
) ENGINE=InnoDB;

-- ---------- USUARIOS ----------
CREATE TABLE usuarios (
  id_usuario      INT NOT NULL AUTO_INCREMENT,
  cedula          VARCHAR(20)  NOT NULL,
  nombre_completo VARCHAR(100) NOT NULL,
  nombre_usuario  VARCHAR(50)  NOT NULL,
  clave_hash      VARCHAR(255) NOT NULL,
  id_rol          INT NOT NULL,
  estado          ENUM('Activo','Inactivo') DEFAULT 'Activo',
  fecha_creacion  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id_usuario),
  UNIQUE KEY uq_usuarios_cedula (cedula),
  UNIQUE KEY uq_usuarios_nombre_usuario (nombre_usuario),
  CONSTRAINT fk_usuarios_rol FOREIGN KEY (id_rol) REFERENCES roles (id_rol) ON UPDATE CASCADE
) ENGINE=InnoDB;

-- ---------- FERRETERIAS (clientes; de aquí sale el estado de destino) ----------
CREATE TABLE ferreterias (
  id_ferreteria  INT NOT NULL AUTO_INCREMENT,
  rif            VARCHAR(20)  NOT NULL,
  razon_social   VARCHAR(150) NOT NULL,
  direccion      TEXT NOT NULL,
  estado_region  VARCHAR(50)  NOT NULL,
  fecha_registro TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id_ferreteria),
  UNIQUE KEY uq_ferreterias_rif (rif)
) ENGINE=InnoDB;

-- ---------- FILAS Y ESTIVAS (área de carga + zona Zulia) ----------
CREATE TABLE filas (
  id_fila            INT NOT NULL AUTO_INCREMENT,
  nombre             VARCHAR(50)  NOT NULL,
  estados_referencia VARCHAR(150) NULL,
  zona               VARCHAR(50)  NOT NULL DEFAULT 'Área de carga',
  orden              INT NOT NULL DEFAULT 0,
  PRIMARY KEY (id_fila),
  UNIQUE KEY uq_filas_nombre (nombre)
) ENGINE=InnoDB;

CREATE TABLE estivas (
  id_estiva        INT NOT NULL AUTO_INCREMENT,
  numero_estiva    INT NOT NULL,
  id_fila          INT NOT NULL,
  capacidad_maxima INT NOT NULL DEFAULT 50,
  bultos_actuales  INT NOT NULL DEFAULT 0,
  PRIMARY KEY (id_estiva),
  UNIQUE KEY uq_estivas_fila_numero (id_fila, numero_estiva),
  CONSTRAINT fk_estivas_fila FOREIGN KEY (id_fila) REFERENCES filas (id_fila) ON DELETE CASCADE
) ENGINE=InnoDB;

-- ---------- CARGAS (viajes de camión) ----------
CREATE TABLE cargas (
  id_carga     BIGINT NOT NULL AUTO_INCREMENT,
  camion       VARCHAR(60) NOT NULL,
  chofer       VARCHAR(80) NULL,
  estado       VARCHAR(10) NOT NULL DEFAULT 'ABIERTA',
  fecha_inicio DATETIME NOT NULL,
  fecha_cierre DATETIME NULL,
  id_creador   INT NOT NULL,
  id_cierre    INT NULL,
  PRIMARY KEY (id_carga),
  KEY idx_cargas_estado (estado),
  CONSTRAINT fk_cargas_creador FOREIGN KEY (id_creador) REFERENCES usuarios (id_usuario),
  CONSTRAINT fk_cargas_cierre  FOREIGN KEY (id_cierre)  REFERENCES usuarios (id_usuario)
) ENGINE=InnoDB;

-- ---------- PEDIDOS ----------
CREATE TABLE pedidos (
  id_pedido     INT NOT NULL AUTO_INCREMENT,
  codigo        VARCHAR(30) NOT NULL,
  tipo          VARCHAR(10) NOT NULL DEFAULT 'OUT',
  id_ferreteria INT NOT NULL,
  bultos        INT NOT NULL,
  fecha_chequeo DATETIME NOT NULL,
  id_chequeador INT NOT NULL,
  nota          VARCHAR(255) DEFAULT NULL,
  situacion     VARCHAR(20) NOT NULL DEFAULT 'CHEQUEADO',
  fecha_ubicado DATETIME DEFAULT NULL,
  id_ubicador   INT DEFAULT NULL,
  id_estiva     INT DEFAULT NULL,
  fecha_cargado DATETIME DEFAULT NULL,
  id_carga      BIGINT DEFAULT NULL,
  id_cargador   INT DEFAULT NULL,
  PRIMARY KEY (id_pedido),
  UNIQUE KEY uq_pedidos_codigo (codigo),
  KEY idx_pedidos_chequeador_fecha (id_chequeador, fecha_chequeo),
  KEY idx_pedidos_fecha (fecha_chequeo),
  KEY idx_pedidos_situacion (situacion),
  KEY idx_pedidos_carga (id_carga),
  CONSTRAINT fk_pedidos_ferreteria FOREIGN KEY (id_ferreteria) REFERENCES ferreterias (id_ferreteria),
  CONSTRAINT fk_pedidos_chequeador FOREIGN KEY (id_chequeador) REFERENCES usuarios (id_usuario),
  CONSTRAINT fk_pedidos_ubicador   FOREIGN KEY (id_ubicador)   REFERENCES usuarios (id_usuario),
  CONSTRAINT fk_pedidos_estiva     FOREIGN KEY (id_estiva)     REFERENCES estivas (id_estiva),
  CONSTRAINT fk_pedidos_carga      FOREIGN KEY (id_carga)      REFERENCES cargas (id_carga),
  CONSTRAINT fk_pedidos_cargador   FOREIGN KEY (id_cargador)   REFERENCES usuarios (id_usuario)
) ENGINE=InnoDB;

FROM filas f
